```python
"""Logic thuần của chế độ đội: xếp đội, đếm phiếu, tính điểm đội."""

from dataclasses import dataclass
from random import Random

from app.game.scoring import points


@dataclass(frozen=True)
class TeamResult:
    option: int | None  # đáp án đội chốt; None nếu hòa hoặc không ai bỏ phiếu
    tied: bool
    gained: int


def smallest_team(sizes: list[int]) -> int:
    """Chỉ số đội ít người nhất; bằng nhau thì lấy chỉ số nhỏ hơn."""
    return min(range(len(sizes)), key=lambda i: (sizes[i], i))


def balanced_assignment(player_ids: list[str], count: int, rng: Random) -> dict[str, int]:
    """Xáo thứ tự rồi chia vòng tròn: số người giữa các đội chênh nhau tối đa 1."""
    shuffled = list(player_ids)
    rng.shuffle(shuffled)
    return {player_id: i % count for i, player_id in enumerate(shuffled)}


def tally(options: list[int], option_count: int) -> tuple[list[int], int | None, bool]:
    """(số phiếu từng ô, ô thắng hoặc None, có hòa không)."""
    counts = [0] * option_count
    for option in options:
        counts[option] += 1
    top = max(counts, default=0)
    if top == 0:
        return counts, None, False
    leaders = [i for i, count in enumerate(counts) if count == top]
    if len(leaders) > 1:
        return counts, None, True
    return counts, leaders[0], False


def team_result(votes: list[tuple[int, int]], option_count: int, correct_index: int, clip_ms: int) -> TeamResult:
    """votes: (ô, thời gian hiệu dụng ms) của từng phiếu."""
    _, winner, tied = tally([option for option, _ in votes], option_count)
    if winner is None or winner != correct_index:
        return TeamResult(winner, tied, 0)
    times = [t_ms for option, t_ms in votes if option == winner]
    return TeamResult(winner, False, points(True, round(sum(times) / len(times)), clip_ms))
```

```python
    # ---- chia đội ----

    @property
    def team_mode(self) -> bool:
        return bool(self.teams)

    def set_teams(self, count: int) -> None:
        """0 là chơi thường; 2–4 là số đội. Đội giữ lại giữ tên và thành viên."""
        self._require(Phase.LOBBY)
        if count != 0 and count not in TEAM_COUNTS:
            raise GameError("INVALID_STATE")
        if count == 0:
            self.teams = []
            for player in self.players.values():
                player.team = None
            return
        kept = self.teams[:count]
        self.teams = kept + [Team(name) for name in TEAM_DEFAULT_NAMES[len(kept) : count]]
        for player in self.players.values():
            if player.team is not None and player.team >= count:
                player.team = None
        # Người chưa có đội (mới bật chế độ, hoặc đội vừa bị bỏ) được xếp theo thứ tự vào phòng.
        for player in self.players.values():
            if player.team is None:
                self._place_in_smallest_team(player)

    def shuffle_teams(self) -> None:
        self._require_team_lobby()
        assignment = balanced_assignment(list(self.players), len(self.teams), self.rng)
        for player_id, team in assignment.items():
            self.players[player_id].team = team

    def move_player(self, player_id: str, team: int) -> None:
        self._require_team_lobby()
        player = self.players.get(player_id)
        if player is None or not 0 <= team < len(self.teams):
            raise GameError("INVALID_STATE")
        player.team = team

    def choose_team(self, player_id: str, team: int) -> None:
        """Người chơi tự chuyển mình; luật giống quản trò chuyển người."""
        self.move_player(player_id, team)

    def rename_team(self, team: int, name: str) -> None:
        self._require_team_lobby()
        if not 0 <= team < len(self.teams):
            raise GameError("INVALID_STATE")
        clean = name.strip()
        if not 1 <= len(clean) <= NAME_MAX_LEN:
            raise GameError("INVALID_NAME")
        self.teams[team].name = clean

    def team_leaderboard(self) -> list[tuple[int, Team, int]]:
        """(chỉ số đội, đội, hạng), sắp theo hạng rồi chỉ số đội."""
        ranks = competition_ranks({index: team.score for index, team in enumerate(self.teams)})
        rows = [(index, team, ranks[index]) for index, team in enumerate(self.teams)]
        return sorted(rows, key=lambda row: (row[2], row[0]))
```

```python
    def _maybe_reveal_early(self) -> None:
        """Sang REVEAL khi mọi người online đã trả lời (chỉ khi có ít nhất 1 người online).

        Chế độ đội: còn đội đang hòa phiếu thì chờ tới hạn chót để đội tự gỡ.
        """
        if self.phase is not Phase.ANSWERING:
            return
        online = [pid for pid, p in self.players.items() if p.online]
        if not online or not all(pid in self.round.answers for pid in online):
            return
        if self.team_mode and any(result.tied for result in self._team_results().values()):
            return
        self._reveal()

    def _reveal(self) -> None:
        for player_id, answer in self.round.answers.items():
            self.players[player_id].score += answer.gained
        if self.team_mode:
            self.round.team_results = self._team_results()
            for team, result in self.round.team_results.items():
                self.teams[team].score += result.gained
        self.phase = Phase.REVEAL

    def _team_results(self) -> dict[int, TeamResult]:
        """Kết quả mọi đội từ phiếu của thành viên hiện tại, kể cả người đang offline."""
        rnd = self.round
        question = rnd.question
        delays = self.effective_delays()
        votes: dict[int, list[tuple[int, int]]] = {index: [] for index in range(len(self.teams))}
        for player_id, answer in rnd.answers.items():
            t_ms = effective_time_ms(answer.received_at, rnd.t0, delays[player_id])
            votes[self.players[player_id].team].append((answer.option_index, t_ms))
        return {
            team: team_result(team_votes, len(question.options), question.correct_index, question.clip_ms)
            for team, team_votes in votes.items()
        }
```

```python
        """Chơi thường: bấm là chốt. Chế độ đội: bỏ hoặc đổi phiếu tới hạn chót."""
        self._require(Phase.ANSWERING)

        rnd = self.round
        if not 0 <= option_index < len(rnd.question.options):
            raise GameError("INVALID_STATE")
        previous = rnd.answers.get(player_id)
        if previous is not None and not self.team_mode:
            raise GameError("ALREADY_ANSWERED")
        if now > rnd.deadline:
            raise GameError("TOO_LATE")
        if previous is not None and previous.option_index == option_index:
            return  # bấm lại ô đang bầu: giữ thời điểm cũ để không bị tính chậm

        rnd.answers[player_id] = Answer(option_index, now, gained)
        self._maybe_reveal_early()
```

```python
def _teams(room: Room, *, by_id: bool) -> list[dict] | None:
    """Theo chỉ số đội. Màn quản trò nhận id thành viên, điện thoại nhận tên."""
    if not room.team_mode:
        return None
    ranks = {index: rank for index, _, rank in room.team_leaderboard()}
    results = room.round.team_results if room.phase is Phase.REVEAL else {}
    rows = []
    for index, team in enumerate(room.teams):
        members = [player for player in room.players.values() if player.team == index]
        row = {
            "index": index,
            "name": team.name,
            "score": team.score,
            "rank": ranks[index],
            "gained": results[index].gained if index in results else None,
        }
        if by_id:
            row["memberIds"] = [player.id for player in members]
        else:
            row["members"] = [player.name for player in members]
        rows.append(row)
    return rows


def _team_picks(room: Room) -> list[dict] | None:

    return [
```

```python
def _team_votes(room: Room, player_id: str) -> list[int] | None:
    """Số phiếu từng ô của đội người nhận; chỉ khi đang trả lời hoặc công bố."""
    team = room.players[player_id].team

        return None
    options = [
        answer.option_index for pid, answer in room.round.answers.items() if room.players[pid].team == team
    ]
    counts, _, _ = tally(options, len(room.round.question.options))
    return counts


def _team_result(room: Room, player_id: str) -> dict | None:
    """Kết quả đội của người nhận; chỉ gọi lúc REVEAL."""

    result = room.round.team_results[team]
    return {
        "option": result.option,
        "tied": result.tied,
        "correct": result.option == room.round.question.correct_index,
        "gained": result.gained,
    }
```

```ts
// face: nền và chữ; swatch: vạch màu; hard: màu bóng cứng, khác màu nền để khối nổi trên nền đen.
// Chuỗi class viết nguyên văn để Tailwind quét thấy.
export const TEAM_COLORS = [
  { face: 'bg-pop-red text-pop-paper', swatch: 'bg-pop-red', hard: 'hard-pop-paper' },
  { face: 'bg-pop-lime text-pop-ink', swatch: 'bg-pop-lime', hard: 'hard-pop-red' },
  { face: 'bg-pop-paper text-pop-ink', swatch: 'bg-pop-paper', hard: 'hard-pop-red' },
  { face: 'bg-pop-violet text-pop-paper', swatch: 'bg-pop-violet', hard: 'hard-pop-lime' },
] as const

// Phần chung của HostTeam và PlayerTeam mà bảng xếp hạng và bục vinh quang cần.
export interface TeamRow {
  index: number
  name: string
  score: number
  rank: number
  gained: number | null
}

// Server gửi teams theo chỉ số đội. Hàm này trả bản sao xếp theo hạng, cùng hạng thì theo chỉ số đội.
export function byRank<T extends TeamRow>(teams: T[]): T[] {
  return [...teams].sort((a, b) => a.rank - b.rank || a.index - b.index)
}

export function teamPodium(teams: TeamRow[]): PodiumEntry[] {
  return byRank(teams).map((team) => ({ id: `team-${team.index}`, name: team.name, score: team.score, rank: team.rank }))
}

// Đang hòa: từ hai ô trở lên cùng bằng số phiếu cao nhất, và số đó lớn hơn 0.
export function votesTied(votes: number[]): boolean {
  const top = Math.max(0, ...votes)
  return top > 0 && votes.filter((count) => count === top).length > 1
}
```

```tsx
// Bấm chip để đổi tên: Enter hoặc rời ô thì lưu, Esc thì hủy.
function TeamName({ team, run }: { team: HostTeam; run: HostRun }) {
  const [editing, setEditing] = useState(false)
  // Enter/Esc làm ô nhập biến mất, có thể kéo theo blur: chỉ xử lý lần kết thúc đầu tiên.
  const done = useRef(false)

  const finish = (value: string | null) => {
    if (done.current) return
    done.current = true
    setEditing(false)
    const name = value?.trim()
    if (name && name !== team.name) void run('host:rename_team', { team: team.index, name })
  }

  if (!editing) {
    return (
      <button
        type="button"
        aria-label={`Đổi tên ${team.name}`}
        onClick={() => {
          done.current = false
          setEditing(true)
        }}
        className="w-fit max-w-full text-left"
      >
        <TeamChip team={team.index} name={team.name} />
      </button>
    )
  }
  return (
    <input
      autoFocus
      defaultValue={team.name}
      maxLength={NAME_MAX_LENGTH}
      aria-label={`Tên ${team.name}`}
      onBlur={(e) => finish(e.currentTarget.value)}
      onKeyDown={(e) => {
        if (e.key === 'Enter') finish(e.currentTarget.value)
        if (e.key === 'Escape') finish(null)
      }}
      className={`w-full max-w-60 -skew-x-12 border-2 border-pop-ink px-3 py-1 text-2xl font-black italic ${TEAM_COLORS[team.index].face}`}
    />
  )
}
```

```tsx
// Chế độ đội (team khác null): ô là phiếu bầu, bấm ô khác để đổi phiếu tới hết giờ.
export function PlayerAnswering({
  round,
  offset,
  team = null,
}: {
  round: PlayerRound
  offset: number
  team?: PlayerTeam | null
}) {
  const now = useNow()
  const deadline = round.deadline ?? 0
  const [totalMs] = useState(() => remainingMs(deadline, offset))
  const [chosen, setChosen] = useState<number | null>(null)
  const [error, setError] = useState<string | null>(null)
  // Chơi thường: câu trả lời server đã nhận là chốt. Chế độ đội: lựa chọn mới nhất trên máy hiện ngay.
  const picked = team ? (chosen ?? round.myAnswer) : (round.myAnswer ?? chosen)

  const choose = async (optionIndex: number) => {
    if (optionIndex === picked) return
    setChosen(optionIndex)
    setError(null)
    const response = await emitAck('player:answer', { roundIndex: round.index, optionIndex })
    if (response.ok || response.error === 'ALREADY_ANSWERED') return
    setChosen(null)
    setError(errorMessage(response.error))
  }

  const tileState = (i: number): TileState =>
    i === picked ? 'chosen' : picked === null || team ? 'idle' : 'dimmed'

  return (
    <section className="flex flex-1 flex-col gap-4">
      <div className="flex items-center justify-between">
        <Title round={round} />
        <CdDisc size="sm" remainingMs={remainingMs(deadline, offset, now)} totalMs={totalMs} />
      </div>
      {/* Cùng vị trí với màn stream: A trên-trái, B trên-phải, C dưới-trái, D dưới-phải. */}
      <div className="grid min-h-[55dvh] flex-1 grid-cols-2 grid-rows-2 gap-3 text-xl">
        {(round.options ?? []).map((label, i) => (
          <AnswerTile
            key={i}
            index={i}
            label={label}
            layout="stack"
            state={tileState(i)}
            votes={round.teamVotes?.[i]}
            disabled={!team && picked !== null}
            onSelect={(index) => void choose(index)}
          />
        ))}
      </div>
      {team ? (
        <TeamStatus team={team} votes={round.teamVotes ?? []} />
      ) : (
        picked !== null && (
          <p className="text-center text-xl font-black text-pop-lime italic">
            Đã chọn {TILES[picked].letter}, chờ kết quả
          </p>
        )
      )}
      {error && <Callout tone="error">{error}</Callout>}
    </section>
  )
}

function TeamStatus({ team, votes }: { team: PlayerTeam; votes: number[] }) {

      >
        ĐANG HÒA!
      </p>
    )

  const sum = votes.reduce((total, count) => total + count, 0)
  return (
    <p className="text-center text-xl font-black text-pop-lime italic">
      {team.name} · {sum}/{team.members.length} đã bỏ phiếu
```

```tsx
const BANNERS = {
  lime: 'bg-pop-lime text-pop-ink shadow-[5px_5px_0_var(--color-pop-red)]',
  paper: 'bg-pop-paper text-pop-ink shadow-[5px_5px_0_var(--color-pop-red)]',
  red: 'bg-pop-red text-pop-paper shadow-[5px_5px_0_var(--color-pop-paper)]',
}

// Chế độ đội báo theo đáp án của đội; chơi thường báo theo câu trả lời của mình.
function revealBanner(result: PlayerResult, myAnswer: number | null, rank: number) {
  const { team } = result
  if (team) {
    if (team.correct) return { line: `ĐỘI ĐÚNG! +${team.gained}`, tone: BANNERS.lime, won: true }
    if (team.tied) return { line: 'HÒA PHIẾU · 0 ĐIỂM', tone: BANNERS.red, won: false }
    return { line: 'ĐỘI SAI', tone: BANNERS.paper, won: false }
  }
  const answer = `${TILES[result.correctIndex].letter} · ${result.animeName}`
  if (result.correct) return { line: `✅ Đúng! +${result.gained} · Hạng ${rank}`, tone: BANNERS.lime, won: true }
  if (myAnswer === null) return { line: `⌛ Hết giờ, đáp án là ${answer}`, tone: BANNERS.paper, won: false }
  return { line: `❌ Sai, đáp án là ${answer}`, tone: BANNERS.red, won: false }
}

export function PlayerReveal({ snapshot, round }: { snapshot: PlayerSnapshot; round: PlayerRound }) {
  const { result, myAnswer } = round
  if (!result) return null
  const { line, tone, won } = revealBanner(result, myAnswer, snapshot.me.rank)
  // Chế độ đội: dấu ✕ đặt trên ô đội chốt.
  const wrongPick = result.team ? result.team.option : myAnswer

  const tileState = (i: number): TileState =>
    i === result.correctIndex ? 'correct' : i === wrongPick ? 'wrong' : 'dimmed'

  return (
    <section className="flex flex-1 flex-col gap-4">
      <Title round={round} />
      <div className="relative">
        {won && <Burst className="absolute -top-10 -right-4 size-32 fill-pop-red motion-safe:animate-slam" />}
        <p className={`relative -skew-x-6 px-4 py-3 text-3xl font-black italic motion-safe:animate-slam ${tone}`}>
          {line}
        </p>
      </div>
      {result.songLabel && (
        <PopChip as="p" className="text-lg">
          {result.songLabel}
        </PopChip>
      )}
      <div className="grid min-h-[45dvh] flex-1 grid-cols-2 grid-rows-2 gap-3 text-xl">
        {(round.options ?? []).map((label, i) => (
          <AnswerTile key={i} index={i} label={label} layout="stack" state={tileState(i)} />
        ))}
      </div>
    </section>
  )
}
```
